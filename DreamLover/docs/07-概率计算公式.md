## 1. 核心公式 -- GetStartRelationSuccessRate_SexRelationBaseRate

源码位置: `GameData.decompiled.cs` 第 558869-559046 行

完整伪代码（逐行对应源码）:

```
FUNCTION SexRelationBaseRate(selfChar, targetChar, selfToTarget, targetToSelf):
    // 第558871行: 基础值
    result = 100

    // 第558872-558875行: 读取性别信息
    selfGender          = selfChar.GetGender()            // 生理性别 (0=女, 1=男)
    selfDisplayGender   = selfChar.GetDisplayingGender()  // 外表性别
    targetGender        = targetChar.GetGender()
    targetDisplayGender = targetChar.GetDisplayingGender()

    // 第558878-558879行: 读取取向与同性判断
    targetBisexual = targetChar.GetBisexual()             // 目标是否双性恋
    sameGender     = (selfGender == targetGender)         // 生理性别是否相同
    sameDisplay    = (selfDisplayGender == targetDisplayGender) // 外表性别是否相同

    //=== 第一关: 性别/取向判定 (第558881-558910行) ===

    // 第558881行: 完全阻止条件
    IF (targetBisexual AND NOT sameGender AND NOT sameDisplay)
       OR (NOT targetBisexual AND sameGender AND sameDisplay):
        RETURN int.MinValue                               // 完全阻止，不可能恋爱

    // 第558889行: 非双性恋 + 生理同性 + 外表异性 => -20
    IF NOT targetBisexual AND sameGender AND NOT sameDisplay:
        result -= 20

    // 第558897行: 双性恋 + 生理异性 + 外表同性 => -20
    IF targetBisexual AND NOT sameGender AND sameDisplay:
        result -= 20

    // 第558906行: 双方都是双性恋 + 生理同性 => +20
    IF sameGender AND selfChar.GetBisexual() AND targetChar.GetBisexual():
        result += 20

    //=== 第二关: 已有关系修正 (第558913-558932行) ===

    // 第558913行: 目标已是自己配偶 => +40
    IF RelationType.HasRelation(targetToSelf.RelationType, 1024):  // 1024 = HusbandOrWife
        result += 40

    // 第558918行: 直系血亲 => -160
    //   ContainDirectBloodRelations 检测 bit mask 0x0007:
    //   BloodParent(1) | BloodChild(2) | BloodBrotherOrSister(4)
    IF RelationType.ContainDirectBloodRelations(targetToSelf.RelationType):
        result -= 160

    // 第558923行: 非血亲家庭关系 => -80
    //   ContainNonBloodFamilyRelations 检测 bit mask 0x03F8:
    //   StepParent(8) | StepChild(16) | StepBrotherOrSister(32)
    //   | AdoptiveParent(64) | AdoptiveChild(128) | AdoptiveBrotherOrSister(256)
    IF RelationType.ContainNonBloodFamilyRelations(targetToSelf.RelationType):
        result -= 80

    // 第558928行: 师徒关系 => -40
    //   2048 = Mentor, 4096 = Mentee
    IF RelationType.HasRelation(selfToTarget.RelationType, 2048)
       OR RelationType.HasRelation(selfToTarget.RelationType, 4096):
        result -= 40

    //=== 第三关: 年龄差惩罚 (第558935-558948行) ===

    selfAge   = selfChar.GetCurrAge()
    targetAge = targetChar.GetCurrAge()

    // 第558937行: 同性情况
    IF selfChar.GetGender() == targetChar.GetGender():
        agePenalty = Clamp(3 * Abs(selfAge - targetAge), 0, 60)
        result -= agePenalty

    // 第558945行: 异性情况 -- 取决于谁是男方
    ELSE:
        IF selfGender == 1:   // self 是男
            IF selfAge > targetAge:
                // 男方年长: 超过10岁部分才扣分
                agePenalty = Clamp(3 * (selfAge - targetAge - 10), 0, 60)
            ELSE:
                // 女方年长: 直接按差值扣分
                agePenalty = Clamp(3 * (targetAge - selfAge), 0, 60)
        ELSE:                 // self 是女
            IF targetAge > selfAge:
                // 男方(target)年长: 超过10岁部分才扣分
                agePenalty = Clamp(3 * (targetAge - selfAge - 10), 0, 60)
            ELSE:
                // 女方(self)年长: 直接按差值扣分
                agePenalty = Clamp(3 * (selfAge - targetAge), 0, 60)
        result -= agePenalty

    //=== 第四关: 好感度加成 (第558951-558957行) ===

    // GetFavorabilityType 将原始好感值映射为 -6 ~ +6 的等级
    //   好感值范围:  -30000 ~ +30000
    //   [-5999, +5999] => type 0 (陌生)
    //   [+6000, +9999] => type 1
    //   [+10000, +13999] => type 2
    //   [+14000, +17999] => type 3 (热忱)
    //   [+18000, +21999] => type 4 (喜爱)
    //   [+22000, +25999] => type 5 (亲密)
    //   [+26000, +30000] => type 6 (不渝)
    //   负面对称

    favType = FavorabilityType.GetFavorabilityType(targetToSelf.Favorability)
    IF favType >= 3:
        result += (favType - 3) * 60
        // type 3 => +0, type 4 => +60, type 5 => +120, type 6 => +180

    //=== 第五关: 魅力差加成 (第558960-558964行) ===

    // GetAttractionType: attraction / 100, 夹紧到 [0, 8]
    //   0=非人, 1=丑陋, 2=不扬, 3=寻常, 4=出众, 5=俊秀, 6=绝伦, 7=绝世, 8=天人
    //   魅力原始值范围: 0 ~ 900
    selfAttrType   = AttractionType.GetAttractionType(selfChar.GetAttraction())
    targetAttrType = AttractionType.GetAttractionType(targetChar.GetAttraction())
    result += (selfAttrType - targetAttrType) * 10
    // self 比 target 好看 => 正加成; target 更好看 => 负修正

    //=== 第六关: 阶层差修正 (第558967-558969行) ===

    // GetInteractionGrade: 太吾取相书等级(0-8), NPC取组织阶级
    selfGrade   = selfChar.GetInteractionGrade(targetChar)
    targetGrade = targetChar.GetInteractionGrade(selfChar)
    result += (selfGrade - targetGrade) * 10
    // self 阶层更高 => 正加成

    //=== 第七关: 外表性别加成 (第558972-558976行) ===

    IF selfChar.GetDisplayingGender() == 0:  // 外表为女性
        result += 20

    //=== 第八关: 非太吾加成 (第558978-558982行) ===

    taiwuId = DomainManager.Taiwu.GetTaiwuCharId()
    IF selfChar.GetId() != taiwuId AND targetChar.GetId() != taiwuId:
        result += 20                          // NPC之间更容易恋爱

    //=== 第九关: 已有爱慕者惩罚 (第558984-559001行) ===

    // 统计目标的存活爱慕者数量(不含self)
    adored = targetChar 的所有爱慕者集合
    aliveAdorerCount = 0
    FOR EACH adorerId IN adored:
        IF adorerId 存活 AND adorerId != selfChar.GetId():
            aliveAdorerCount++

    IF aliveAdorerCount > 0:
        // CalcAdoreMultiplePeopleChanceFactor: 遍历目标的所有特征
        //   统计 AdoreMultiplePeopleChanceFactor != 100 的特征
        //   返回这些特征的 AdoreMultiplePeopleChanceFactor 平均值
        //   如果没有相关特征则返回 100
        //
        //   已知特征值:
        //     Feature 162 (情有独钟): 80    -- 正面一级
        //     Feature 163 (从一而终): 60    -- 正面二级
        //     Feature 164 (忠贞不渝): 40    -- 正面三级
        //     Feature 165 (三心二意): 120   -- 负面一级
        //     Feature 166 (见异思迁): 140   -- 负面二级
        //     Feature 167 (朝三暮四): 160   -- 负面三级
        featureFactor = CalcAdoreMultiplePeopleChanceFactor(targetChar)

        // StartAdoreChanceChangePerAdored 按行为类型索引:
        //   [0]=刚正: -50
        //   [1]=仁善: -40
        //   [2]=中庸: -30
        //   [3]=叛逆: -10
        //   [4]=唯我: -20
        perAdoredChange = StartAdoreChanceChangePerAdored[targetChar.GetBehaviorType()]

        adorerPenalty = aliveAdorerCount * perAdoredChange * featureFactor / 100
        result += adorerPenalty
        // 注: perAdoredChange 为负数, 所以 adorerPenalty 通常为负
        // featureFactor > 100 (三心二意等) 放大惩罚 => 反而更易恋爱（因为惩罚是负*负=正？）
        // 实际上 perAdoredChange 是负的, featureFactor > 100 使结果更负 => 更难恋爱
        // featureFactor < 100 (情有独钟等) 缩小惩罚绝对值 => 更容易恋爱

    //=== 第十关: Feature 685 阻止 (第559003-559011行) ===

    // 685 = "梦境中人" (太吾挚友专属特征)
    IF targetChar 拥有 Feature 685
       AND selfChar.GetId() != 太吾挚友ID:
        RETURN int.MinValue                   // 完全阻止

    //=== 第十一关: 子嗣数量惩罚 (第559013-559019行) ===

    selfChildren   = selfChar 的亲生子女数 + 继子女数
    targetChildren = targetChar 的亲生子女数 + 继子女数
    totalChildren  = selfChildren + targetChildren
    result -= 10 * totalChildren

    //=== 第十二关: 出家惩罚 (第559022-559031行) ===

    // MonkType 常量:
    //   0     = 未出家
    //   1     = 非门派道士 (NonSectTaoist)
    //   2     = 非门派佛门 (NonSectBuddhist)
    //   129   = 门派道士 (SectTaoist, 128|1)
    //   130   = 门派佛门 (SectBuddhist, 128|2)
    IF selfChar.GetMonkType() != 0:
        result -= 80

    IF targetChar.GetMonkType() != 0:
        result -= 80
    // 双方都出家则 -160

    //=== 第十三关: 生育力惩罚 (第559034-559043行) ===

    IF selfChar.GetFertility() <= 0:
        result -= 30

    IF targetChar.GetFertility() <= 0:
        result -= 30
    // 双方都无生育力则 -60

    RETURN result
```

FUNCTION Adored_SuccessRate(selfChar, targetChar, selfToTarget, targetToSelf):
    // 第558789行: 注意参数颠倒! 是 target 对 self 的基础值
    rate = SexRelationBaseRate(targetChar, selfChar, targetToSelf, selfToTarget)

    // 第558790行: 如果被完全阻止
    IF rate == int.MinValue:
        RETURN 0

    // 第558794行: 直系血亲近亲系数
    //   ContainDirectBloodRelations 检测 selfToTarget 的关系
    IF RelationType.ContainDirectBloodRelations(selfToTarget.RelationType):
        behaviorType = selfChar.GetBehaviorType()
        // IncestSexRelationChanceFactor 按行为类型索引:
        //   [0]=刚正:  5   => rate * 5 / 100  = rate * 0.05
        //   [1]=仁善: 10   => rate * 10 / 100 = rate * 0.10
        //   [2]=中庸: 15   => rate * 15 / 100 = rate * 0.15
        //   [3]=叛逆: 25   => rate * 25 / 100 = rate * 0.25
        //   [4]=唯我: 20   => rate * 20 / 100 = rate * 0.20
        rate = rate * IncestSexRelationChanceFactor[behaviorType] / 100

    RETURN rate

FUNCTION BoyOrGirlFriend_SuccessRate(selfChar, targetChar, selfToTarget, targetToSelf):
    // 第558821行: 计算基础值 (self -> target 方向)
    rate = SexRelationBaseRate(selfChar, targetChar, selfToTarget, targetToSelf)

    IF rate == int.MinValue:
        RETURN 0

    // 第558826行: 表白基础惩罚
    rate -= 20

    // 第558827-558838行: 配偶惩罚
    selfSpouse   = GetAliveSpouse(selfChar)
    targetSpouse = GetAliveSpouse(targetChar)

    IF selfSpouse 存在 AND selfSpouse != targetChar.GetId():
        rate -= 60              // 自己有别的配偶
    IF targetSpouse 存在 AND targetSpouse != selfChar.GetId():
        rate -= 60              // 目标有别的配偶

    // 第558840行: AI 行动成功率调整 (来自事件/技能等外部修正)
    rate += GetAiActionSuccessRateAdjust(selfChar.Id, targetChar.Id,
                actionType=10, actionSubType=15)
    // actionSubType = RelationType.GetTypeId(16384) = GetTypeId(Adored) = 15

    // 第558841-558845行: 直系血亲近亲系数 (基于 target 的行为类型)
    IF RelationType.ContainDirectBloodRelations(targetToSelf.RelationType):
        behaviorType = targetChar.GetBehaviorType()
        rate = rate * IncestSexRelationChanceFactor[behaviorType] / 100

    // 第558846-558848行: 太吾表白加成
    IF selfChar.IsTaiwu():
        rate *= GetTaiwuPropertyPermanentBonus(ConfessingLoveSuccessOdds)
        // 此乘数来自太吾的永久属性加成

    RETURN rate

## 5. AI 发起/终止关系概率 -- GetStartOrEndRelationChance

源码位置: `GameData.decompiled.cs` 第 558706-558785 行

```
FUNCTION GetStartOrEndRelationChance(aiRelationsCfg, selfChar, targetChar,
                                      curRelationType, sectFavorability,
                                      multiplier=1):
    behaviorType  = selfChar.GetBehaviorType()     // 0-4
    behaviorType2 = targetChar.GetBehaviorType()
    fameType      = selfChar.GetFameType()          // 0-6
    fameType2     = targetChar.GetFameType()

    adjust = 0

    //=== 行为类型非矛盾调整 (第558713-558716行) ===
    // BehaviorType.IsContradictory 矛盾表:
    //   刚正(0): 矛盾于 [叛逆(3), 唯我(4)]
    //   仁善(1): 矛盾于 [叛逆(3), 唯我(4)]
    //   中庸(2): 矛盾于 [刚正(0), 唯我(4)]
    //   叛逆(3): 矛盾于 [刚正(0), 仁善(1)]
    //   唯我(4): 矛盾于 [刚正(0), 仁善(1)]
    IF aiRelationsCfg.NoncontradictoryBehaviorAjust != 0
       AND NOT BehaviorType.IsContradictory(behaviorType, behaviorType2):
        adjust += aiRelationsCfg.NoncontradictoryBehaviorAjust

    //=== 声望类型非矛盾调整 (第558717-558719行) ===
    IF aiRelationsCfg.NoncontradictoryFameAjust != 0
       AND NOT FameType.IsContradictory(fameType, fameType2):
        adjust += aiRelationsCfg.NoncontradictoryFameAjust

    //=== 门派好感调整 (第558721-558728行) ===
    IF aiRelationsCfg.EnemySectMemberAdjust != 0 AND sectFavorability == -1:
        adjust += aiRelationsCfg.EnemySectMemberAdjust    // 敌对门派
    ELSE IF aiRelationsCfg.FriendlySectMemberAdjust != 0 AND sectFavorability == 1:
        adjust += aiRelationsCfg.FriendlySectMemberAdjust // 友好门派

    //=== 模板特定逻辑 (第558731-558781行) ===
    chanceMul = 100   // 概率乘数百分比

    SWITCH aiRelationsCfg.TemplateId:

        CASE 0:  // 仇敌 (StartEnemy)
            // 第558734行: Feature 741 = "恶体转世"
            IF selfChar 拥有 Feature 741 OR targetChar 拥有 Feature 741:
                chanceMul *= 10          // 概率 x10（更易仇视）

        CASE 2:  // 爱慕 (StartAdored)
            // 第558741-558746行: 已有爱慕者的惩罚
            adoredCount = selfChar 的爱慕者数量
            IF adoredCount > 0:
                perAdoredChange = StartAdoreChanceChangePerAdored[behaviorType]
                featureFactor   = CalcAdoreMultiplePeopleChanceFactor(selfChar)
                chanceMul += adoredCount * perAdoredChange * featureFactor / 100

            // 第558748行: Feature 740 = "净体转世"
            IF selfChar 拥有 Feature 740 OR targetChar 拥有 Feature 740:
                chanceMul *= 10          // 概率 x10（更易爱慕）

            // 第558752行: Feature 685 = "梦境中人"
            IF selfChar 拥有 Feature 685 AND targetChar != 太吾挚友:
                RETURN 0                 // 完全阻止

        CASE 3:  // 表白 (StartBoyOrGirlFriend)
            // 第558759行: Goal 251 + 关系类型 Adored (16384)
            IF selfChar 拥有 Goal(251, 16384):
                chanceMul = chanceMul * 150 / 100  // 概率 x1.5

            // 第558763行: Feature 685 阻止
            IF selfChar 拥有 Feature 685 AND targetChar != 太吾挚友:
                RETURN 0

        CASE 4:  // 分手 (EndBoyOrGirlFriend)
            // 第558769行: 每多一个爱慕者, +20%
            chanceMul += 20 * (selfChar.爱慕者数量 - 1)

        CASE 8:  // 结义 (StartSwornBrotherOrSister)
            // 第558772行: 已有结义越多越难
            chanceMul -= 20 * selfChar.结义兄弟姐妹数量

            // 第558773行: Goal 251 + 关系类型 SwornBrotherOrSister (512)
            IF selfChar 拥有 Goal(251, 512):
                chanceMul = chanceMul * 150 / 100

            // 第558777行: 已有 Adored 关系 => -2000 大幅降低
            IF RelationType.HasRelation(curRelationType, 16384):
                adjust -= 2000

    //=== 基础概率选择 (第558782-558783行) ===
    // 根据当前关系类型选择概率表中的不同列
    IF ContainDirectBloodRelations(curRelationType)
       OR HasRelation(curRelationType, 1024):     // 血亲或配偶
        adjust += aiRelationsCfg.Probability[behaviorType].BloodRelationsProb

    ELSE IF ContainNonBloodFamilyRelations(curRelationType)
            OR HasRelation(curRelationType, 2048)  // Mentor
            OR HasRelation(curRelationType, 4096): // Mentee
        adjust += aiRelationsCfg.Probability[behaviorType].SwornOrAdoptiveRelationsProb

    ELSE:
        adjust += aiRelationsCfg.Probability[behaviorType].DefaultProb

    //=== 最终计算 (第558784行) ===
    RETURN Max(0, adjust * chanceMul * multiplier / 100)
```

## 6. 辅助函数 -- CalcAdoreMultiplePeopleChanceFactor

源码位置: `GameData.decompiled.cs` 第 559048-559066 行

```
FUNCTION CalcAdoreMultiplePeopleChanceFactor(character):
    totalFactor = 0
    count = 0

    FOR EACH featureId IN character.GetFeatureIds():
        factor = CharacterFeature[featureId].AdoreMultiplePeopleChanceFactor
        IF factor != 100:           // 100 表示无修正
            totalFactor += factor
            count++

    IF count != 0:
        RETURN totalFactor / count  // 取平均值
    ELSE:
        RETURN 100                  // 默认无修正
```

FUNCTION GetFavorabilityType(favorability):    // 入参: short, 范围 -30000 ~ +30000
    IF favorability == short.MinValue (-32768):
        RETURN 0                               // 未知 => 视为陌生

    IF favorability >= 6000:
        type = 1 + (favorability - 6000) / 4000
        RETURN Min(type, 6)

    IF favorability <= -6000:
        type = -1 + (favorability + 6000) / 4000
        RETURN Max(type, -6)

    RETURN 0                                   // 陌生
