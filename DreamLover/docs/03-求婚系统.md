## 概述

```pseudocode
FUNCTION CanStartRelation_HusbandOrWife(
    selfCharId, selfToTarget, selfBehaviorType,
    targetCharId, targetToSelf, targetBehaviorType
) -> bool:

    // L558674: 五重前置检查, 任一失败则返回 false
    IF (selfToTarget.RelationType & 0x4000) == 0       // self 未爱慕 target
        RETURN false
    IF (targetToSelf.RelationType & 0x4000) == 0        // target 未爱慕 self (必须双向爱慕)
        RETURN false
    IF ContainBloodExclusionRelations(selfToTarget.RelationType)  // 有血亲排斥关系
        RETURN false
    IF GetAliveSpouse(selfCharId) >= 0                  // self 有存活配偶
        RETURN false
    IF GetAliveSpouse(targetCharId) >= 0                // target 有存活配偶
        RETURN false

    // L558678-558681: 好感度门槛检查
    favorabilityType = FavorabilityType.GetFavorabilityType(selfToTarget.Favorability)
    config = GetAiRelationConfig(5)             // type=5 对应夫妻关系
    minFav = config.MinFavorability[selfBehaviorType]
    RETURN favorabilityType >= minFav
```

**触发条件汇总:**
1. 双方互相爱慕 (都有 Adored 0x4000)
2. 无血亲排斥关系
3. 双方都无存活配偶 (GetAliveSpouse < 0)
4. self 对 target 好感度等级 >= AiRelationConfig[5].MinFavorability[selfBehaviorType]

```pseudocode
    // === 性别兼容性检查 (L558872-558904) ===
    selfGender     = selfChar.GetGender()
    selfDispGender = selfChar.GetDisplayingGender()
    tgtGender      = targetChar.GetGender()
    tgtDispGender  = targetChar.GetDisplayingGender()
    bisexual       = targetChar.GetBisexual()
    sameGender     = (selfGender == tgtGender)
    sameDispGender = (selfDispGender == tgtDispGender)

    // L558881: 完全不兼容 -> 返回 int.MinValue (成功率=0)
    IF (bisexual AND !sameGender AND !sameDispGender)
       OR (!bisexual AND sameGender AND sameDispGender):
        RETURN int.MinValue

    // L558889-558896: 部分兼容惩罚
    IF !bisexual AND sameGender AND !sameDispGender:
        rate -= 20                                      // 实际性别同, 外观性别异
    IF bisexual AND !sameGender AND sameDispGender:
        rate -= 20                                      // 双性恋但外观性别同

    // === 双方同性且都是双性恋 (L558906-558910) ===
    IF sameGender AND selfChar.GetBisexual() AND targetChar.GetBisexual():
        rate += 20

    // === 关系修正 (L558913-558932) ===

        rate += 40                                      // 已有夫妻关系加成
    IF 有直系血亲关系 (ContainDirectBloodRelations):
        rate -= 160                                     // 严重惩罚
    IF 有非血缘家庭关系 (ContainNonBloodFamilyRelations):
        rate -= 80
    IF self 是 target 的师父或徒弟 (0x800 或 0x1000):
        rate -= 40

    // === 年龄差惩罚 (L558935-558948) ===
    selfAge  = selfChar.GetCurrAge()
    tgtAge   = targetChar.GetCurrAge()
    IF 同性:
        penalty = clamp(3 * |selfAge - tgtAge|, 0, 60)
    ELSE:
        // 男方年长 10 岁以内不罚; 超出部分和反向差距都罚
        IF selfGender == 男(1):
            IF selfAge > tgtAge:
                penalty = clamp(3 * (selfAge - tgtAge - 10), 0, 60)
            ELSE:
                penalty = clamp(3 * (tgtAge - selfAge), 0, 60)
        ELSE:
            IF tgtAge > selfAge:
                penalty = clamp(3 * (tgtAge - selfAge - 10), 0, 60)
            ELSE:
                penalty = clamp(3 * (selfAge - tgtAge), 0, 60)
    rate -= penalty

    // === 好感度加成 (L558951-558957) ===

    IF favType >= 3:
        rate += (favType - 3) * 60                      // 好感等级3起, 每级+60

    // === 魅力差值 (L558960-558964) ===

    tgtAttr  = AttractionType.GetAttractionType(targetChar.GetAttraction())

    // === 交互等级差值 (L558967-558969) ===

           - targetChar.GetInteractionGrade(selfChar)) * 10

    // === 外观性别男性加成 (L558972-558976) ===
    IF selfChar.GetDisplayingGender() == 0 (男):
        rate += 20

    // === NPC 对 NPC 加成 (L558978-558982) ===
    taiwuId = DomainManager.Taiwu.GetTaiwuCharId()
    IF self != taiwu AND target != taiwu:
        rate += 20                                      // NPC间互动更容易

    // === 目标已有其他爱慕者惩罚 (L558984-559002) ===
    adoredSet = GetRelatedCharacters(target).Adored.GetCollection()
    otherAdorerCount = count of alive chars in adoredSet excluding self

        rate += otherAdorerCount * perAdored * factor / 100

    // === 特性 685 (专情/心有所属) 排斥 (L559003-559010) ===
    IF target 有特性 685 AND self != 太吾密友:

    selfChildCount  = self.BloodChildren.Count + self.StepChildren.Count
    tgtChildCount   = target.BloodChildren.Count + target.StepChildren.Count
    rate -= 10 * (selfChildCount + tgtChildCount)

    // === 出家惩罚 (L559022-559031) ===
    IF selfChar.GetMonkType() != 0:
        rate -= 80
    IF targetChar.GetMonkType() != 0:
        rate -= 80

        rate -= 30
    IF targetChar.GetFertility() <= 0:
        rate -= 30

    RETURN rate
```

## 7. 远程婚配 -- UpdateDistantMarriages

非太吾队伍的 NPC 自动匹配结婚。条件:
- 可互动的智慧角色 (IsInteractableAsIntelligentCharacter)
- 非太吾队伍、非临时角色
- **非门派成员** (IsSect == false)
- 非出家人 (MonkType == 0)、非双性恋
- 无存活配偶
- 门派允许 (ChildGrade >= 0)、OrgTemplateId != 0 且 != 16
- 据点主事、SettlementId >= 0
- 在常规地区 (MapAreaData.IsRegularArea)
- 目标必须异性、年龄/阶层在范围内、无血亲排斥、无家族仇恨
